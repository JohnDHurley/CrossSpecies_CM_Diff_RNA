RNA_fc <- readRDS("data/QC/concat/RNA_fc.RDS")

total_reads_df <- RNA_fc %>%
  as.data.frame() %>%
  mutate(Gene = rownames(.)) %>%
  dplyr::select(-Gene) %>%
  summarise(across(everything(), \(x) sum(x, na.rm = TRUE))) %>%
  pivot_longer(
    cols = everything(),
    names_to = "Sample",
    values_to = "Total_Counts"
  ) %>%
  left_join(RNA_Metadata, by = c("Sample" = "SampleName"))

day_levels <- c("Day0","Day2","Day4","Day5","Day15","Day30")

total_reads_df <- total_reads_df %>%
  mutate(
    Timepoint = factor(Timepoint, levels = day_levels, ordered = TRUE),

    # force species order: chimp first, human second
    Species = factor(Species, levels = c("C", "H")),

    Individual_Color = ann_colors$individual_cor[Individual]
  ) %>%
  arrange(Species, Individual, Timepoint) %>%
  mutate(Sample = factor(Sample, levels = Sample))

    filter(!is.na(Day), !is.na(Group), !is.na(Percent_Positive)) %>%
    group_by(Day, Marker) %>%
    summarise(
      H_Mean = mean(Percent_Positive[Group == "H"], na.rm = TRUE),
      C_Mean = mean(Percent_Positive[Group == "C"], na.rm = TRUE),
      H_SD = sd(Percent_Positive[Group == "H"], na.rm = TRUE),
      C_SD = sd(Percent_Positive[Group == "C"], na.rm = TRUE),
      H_N = sum(Group == "H"),
      C_N = sum(Group == "C"),
      p_value = t.test(
        Percent_Positive ~ Group,
        data = pick(Percent_Positive, Group),
        var.equal = FALSE
      )$p.value,
      .groups = "drop"
    ) %>%
    group_by(Day) %>%
    mutate(
      p_adj = p.adjust(p_value, method = correction),
      Significance = case_when(
        p_adj < 0.001 ~ "***",
        p_adj < 0.01 ~ "**",
        p_adj < 0.05 ~ "*",
        TRUE ~ "ns"
      )
    ) %>%
    ungroup()

  return(results)
}
