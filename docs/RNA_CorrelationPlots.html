plot_R2_trajectories_No4 <- function(
    R2_table,
    colors
) {

  R2_table <- R2_table %>%
    mutate(
      Group = case_when(
        Gene_Set == "All Expressed Genes" ~ "All Expressed",
        Gene_Set == "All Expressed TFs" ~ "TFs",
        Gene_Set == "All ZNF Genes" ~ "ZNFs",
        Gene_Set == "All Expressed KZFPs" ~ "KZFPs"
      ),
      
      Transition = case_when(
        Transition == "Day02" ~ "Day0-2",
        Transition == "Day25" ~ "Day2-5",
        Transition == "Day515" ~ "Day5-15",
        Transition == "Day1530" ~ "Day15-30"
      )
    )
  # Set factor order
  R2_table$Transition <- factor(
    R2_table$Transition,
    levels = c(
      "Day0-2",
      "Day2-5",
      "Day5-15",
      "Day15-30"
    )
  )

  R2_table$Group <- factor(
    R2_table$Group,
    levels = c(
      "All Expressed",
      "TFs",
      "ZNFs",
      "KZFPs"
    )
  )

  # Plot
  ggplot(
    R2_table,
    aes(
      x = Transition,
      y = R2,
      group = Group,
      linetype = Group
    )
  ) +

    annotate(
      "rect",
      xmin = 1.5,
      xmax = 3.5,
      ymin = -Inf,
      ymax = Inf,
      alpha = 0.1,
      fill = "grey70"
    ) +

    geom_line(
      aes(
        linewidth = Group,
        color = Group
      )
    ) +

# Significant points
geom_point(
  data = R2_table %>% filter(P_value < 0.05),
  aes(color = Group),
